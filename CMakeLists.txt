cmake_minimum_required(VERSION 3.24)
project(curvenet_core VERSION 0.2.1 LANGUAGES CXX)

option(CURVENET_MAYA "Build the Windows Maya 2023 deformer adapter" OFF)

include(CTest)
set(EIGEN_ROOT "" CACHE PATH "Optional path to Eigen 3.4.0 headers")
if(EIGEN_ROOT)
  if(NOT EXISTS "${EIGEN_ROOT}/Eigen/Core")
    message(FATAL_ERROR "EIGEN_ROOT must contain Eigen/Core")
  endif()
  add_library(curvenet_eigen INTERFACE)
  target_include_directories(curvenet_eigen SYSTEM INTERFACE "${EIGEN_ROOT}")
  add_library(Eigen3::Eigen ALIAS curvenet_eigen)
else()
  find_package(Eigen3 3.4 QUIET NO_MODULE)
  if(NOT Eigen3_FOUND)
    include(FetchContent)
    FetchContent_Declare(eigen
      URL https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.zip
      URL_HASH SHA256=eba3f3d414d2f8cba2919c78ec6daab08fc71ba2ba4ae502b7e5d4d99fc02cda
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE
      SOURCE_SUBDIR curvenet_headers_only)
    FetchContent_MakeAvailable(eigen)
    add_library(curvenet_eigen INTERFACE)
    target_include_directories(curvenet_eigen SYSTEM INTERFACE "${eigen_SOURCE_DIR}")
    add_library(Eigen3::Eigen ALIAS curvenet_eigen)
  endif()
endif()

add_library(curvenet_core STATIC src/core.cpp src/rig.cpp)
add_library(CurveNet::core ALIAS curvenet_core)
target_compile_features(curvenet_core PUBLIC cxx_std_17)
target_include_directories(curvenet_core PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_link_libraries(curvenet_core PUBLIC Eigen3::Eigen)
target_compile_definitions(curvenet_core PUBLIC EIGEN_MPL2_ONLY)
if(MSVC)
  target_compile_options(curvenet_core PRIVATE /W4 /permissive- /EHsc /bigobj)
endif()

if(CURVENET_MAYA)
  add_subdirectory(maya)
endif()

add_executable(curvenet_example examples/basic.cpp)
target_link_libraries(curvenet_example PRIVATE CurveNet::core)
if(BUILD_TESTING)
  add_executable(curvenet_tests tests/core_tests.cpp)
  target_link_libraries(curvenet_tests PRIVATE CurveNet::core)
  add_test(NAME curvenet_core COMMAND curvenet_tests)
  add_test(NAME curvenet_example COMMAND curvenet_example)
  find_package(Python3 3.9 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME maya_authoring_data COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/maya/tests/authoring_data_tests.py")
  endif()
endif()
